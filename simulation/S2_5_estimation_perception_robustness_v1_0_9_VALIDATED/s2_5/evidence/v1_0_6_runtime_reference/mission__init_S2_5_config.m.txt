function cfg = init_S2_5_config()
% INIT_S2_5_CONFIG Estimation/perception robustness qualification overlay.
% Inherits the validated S2.4-G/F control, planning, mapping and ESKF values
% unchanged. S2.5 adds only simulation fault-injection metadata and
% qualification thresholds; estimator/controller thresholds are not relaxed.
cfg=init_S2_4_F_config();
cfg.stage='S2.5';
cfg.version='v1.0.6-estimation-perception-robustness-candidate';
% S2.5 sensor/perception qualification must not simultaneously inject the
% S2.4-F execution-grid validation faults.
cfg.executionSafety.validationFault=struct('name','NONE','triggerDelay_s',0.10, ...
    'triggerProgress',0.20,'versionOffset',1000000,'repeatPerAuthority',false);
cfg.s25Robustness=struct( ...
    'enabled',true, ...
    'maxNavigationFaultApplications',100000, ...
    'maxPerceptionFaultApplications',100000, ...
    'requireZeroUnknownCommitment',true, ...
    'requireZeroCollision',true, ...
    'requireZeroGeofence',true, ...
    'requireTruthIsolation',true);
end
